import httpx
import json
import os
import psycopg2
from psycopg2.extras import RealDictCursor

DB_DSN = os.getenv("DATABASE_URL", "postgresql://bam_admin:bam_secure_super_password_2026@postgres:5432/bam_media_factory")
OLLAMA_URL = os.getenv("OLLAMA_URL", "http://host.docker.internal:11434")

def call_ollama(model: str, prompt: str, image_path: str = None):
    payload = {"model": model, "prompt": prompt, "stream": False, "keep_alive": 0, "format": "json"}
    if image_path:
        import base64
        with open(image_path, "rb") as img:
            payload["images"] = [base64.b64encode(img.read()).decode("utf-8")]

    res = httpx.post(f"{OLLAMA_URL}/api/generate", json=payload, timeout=120.0)

    if res.status_code != 200:
        print(f"\n[AI ERROR] Failed to run model '{model}': {res.text}")
        return {}

    return json.loads(res.json().get("response", "{}"))

def execute_phased_batch():
    conn = psycopg2.connect(DB_DSN, cursor_factory=RealDictCursor)
    cur = conn.cursor()

    cur.execute("SELECT * FROM assets WHERE workflow_state = 'INGESTED'")
    assets = cur.fetchall()
    if not assets:
        print("No assets to process.")
        return

    screener_res = {}
    verifier_res = {}

    # Phase 1A: Screener (Llama 3.2 Vision)
    print(f"Running Screener on {len(assets)} assets...")
    for a in assets:
        screener_res[a["asset_id"]] = call_ollama(
            "moondream",
            "Categorize this scene. Is it a drone shot? Return JSON: {'subject': '...', 'perspective': 'nadir or oblique'}",
            a["source_path"]
        )

    # Phase 1B: Verifier (LLaVA)
    print("Running Verifier...")
    for a in assets:
        verifier_res[a["asset_id"]] = call_ollama(
            "llava",
            "Inspect details for engineering. Return JSON: {'subject': '...', 'perspective': 'nadir or oblique'}",
            a["source_path"]
        )

    # Phase 1C: Judge (Llama 3.1)
    print("Running Judge...")
    for a in assets:
        aid = a["asset_id"]
        prompt = f"Compare Screener: {screener_res[aid]} vs Verifier: {verifier_res[aid]}. Return JSON: {{'agreementScore': 0.8}}"
        judge_res = call_ollama("llama3.1", prompt)

        try:
            score = float(judge_res.get("agreementScore", 0.0))
        except (ValueError, TypeError):
            score = 0.0

        cur.execute("""
            INSERT INTO ai_evaluations (asset_id, screener_output, verifier_output, judge_output, agreement_score)
            VALUES (%s, %s, %s, %s, %s)
        """, (aid, json.dumps(screener_res[aid]), json.dumps(verifier_res[aid]), json.dumps(judge_res), score))

        cur.execute("UPDATE assets SET workflow_state = 'PENDING_HUMAN' WHERE asset_id = %s", (aid,))

    conn.commit()
    conn.close()
    print("Batch complete. Awaiting human review.")

if __name__ == "__main__":
    execute_phased_batch()
